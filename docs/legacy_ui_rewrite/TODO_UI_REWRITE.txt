```lua
local function MakeDetailTab(parent, text, x)
  local tab = CreateFrame("Button", nil, parent)
  tab:SetSize(70, 24)
  tab:SetPoint("LEFT", parent, "LEFT", x, 0)
  
  local label = tab:CreateFontString(nil, "OVERLAY", "GameFontNormal")
  label:SetPoint("CENTER")
  label:SetText(text)
  
  local underline = tab:CreateTexture(nil, "OVERLAY")
  underline:SetHeight(2)
  underline:SetPoint("BOTTOMLEFT", 2, 0)
  underline:SetPoint("BOTTOMRIGHT", -2, 0)
  underline:SetColorTexture(0.35, 0.55, 0.85, 1) -- accent
  underline:Hide()
  tab._underline = underline
  
  function tab:SetActive(isActive)
    underline:SetShown(isActive)
    if isActive then
      label:SetTextColor(1, 1, 1)
    else
      label:SetTextColor(0.6, 0.6, 0.6)
    end
  end
  
  tab:SetScript("OnEnter", function()
    label:SetTextColor(1, 1, 1)
  end)
  tab:SetScript("OnLeave", function()
    if not underline:IsShown() then
      label:SetTextColor(0.6, 0.6, 0.6)
    end
  end)
  
  return tab
end
```
